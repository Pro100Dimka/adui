"""Nine weights of Melodix from the large (90 px) trace: each glyph is drawn big, its strokes
grown or slimmed by an exact distance, and traced again. Hairlines and small ornaments are
kept at their own width when slimming, so notes, keys and records survive in every weight."""
import os
import numpy as np
import potrace
from scipy import ndimage
from PIL import Image, ImageDraw
from fontTools.ttLib import TTFont
from fontTools.pens.recordingPen import RecordingPen, DecomposingRecordingPen
from fontTools.pens.transformPen import TransformPen
from fontTools.pens.ttGlyphPen import TTGlyphPen
from fontTools.pens.cu2quPen import Cu2QuPen
from fontTools.pens.reverseContourPen import ReverseContourPen
from fontTools.pens.areaPen import AreaPen
from fontTools.pens.basePen import BasePen
from fontTools import subset

SRC = "Melodix-Regular.ttf"          # the trace of the large sheet
PX = 0.9                             # pixels per font unit while morphing (cap 700 → 630 px)
OFFSETS = {100: -12, 200: -9, 300: -6, 400: -3, 500: 7, 600: 11, 700: 16, 800: 21, 900: 26}
# light weights: half-width (font units) every band and hairline is brought down to, so the gaps
# between the many fine bands open up instead of greying into one blot at text sizes
HAIR = {100: 2.0, 200: 2.9, 300: 3.8, 400: 5.0}
NAMES = {100: "Thin", 200: "ExtraLight", 300: "Light", 400: "Regular", 500: "Medium", 600: "SemiBold", 700: "Bold", 800: "ExtraBold", 900: "Black"}


class PolyPen(BasePen):
    """Flattens a glyph into polygons for drawing with PIL."""
    def __init__(self, gs, tx):
        super().__init__(gs)
        self.polys, self.cur, self.tx = [], [], tx
    def _moveTo(self, p):
        self.cur = [self.tx(p)]
    def _lineTo(self, p):
        self.cur.append(self.tx(p))
    def _curveToOne(self, a, b, c):
        p0 = self._getCurrentPoint()
        for i in range(1, 9):
            t = i / 8
            mt = 1 - t
            x = mt**3 * p0[0] + 3 * mt * mt * t * a[0] + 3 * mt * t * t * b[0] + t**3 * c[0]
            y = mt**3 * p0[1] + 3 * mt * mt * t * a[1] + 3 * mt * t * t * b[1] + t**3 * c[1]
            self.cur.append(self.tx((x, y)))
    def _qCurveToOne(self, a, c):
        p0 = self._getCurrentPoint()
        for i in range(1, 7):
            t = i / 6
            mt = 1 - t
            self.cur.append(self.tx((mt * mt * p0[0] + 2 * mt * t * a[0] + t * t * c[0], mt * mt * p0[1] + 2 * mt * t * a[1] + t * t * c[1])))
    def _closePath(self):
        if len(self.cur) > 2:
            self.polys.append(self.cur)
        self.cur = []
    _endPath = _closePath


def raster(gs, name, pad, extra=None):
    g = gs[name]
    rec = DecomposingRecordingPen(gs)
    g.draw(rec)
    from fontTools.pens.boundsPen import BoundsPen
    bp = BoundsPen(gs)
    g.draw(bp)
    if not bp.bounds:
        return None
    x0, y0, x1, y1 = bp.bounds
    if extra:
        x0, y0, x1, y1 = min(x0, extra[0]), min(y0, extra[1]), max(x1, extra[2]), max(y1, extra[3])
    W = int((x1 - x0) * PX) + 2 * pad
    H = int((y1 - y0) * PX) + 2 * pad
    tx = lambda p: ((p[0] - x0) * PX + pad, (y1 - p[1]) * PX + pad)
    pen = PolyPen(gs, tx)
    rec.replay(pen)
    # even-odd fill reproduces holes regardless of contour direction
    mask = np.zeros((H, W), bool)
    for poly in pen.polys:
        im = Image.new("1", (W, H), 0)
        ImageDraw.Draw(im).polygon(poly, fill=1)
        mask ^= np.asarray(im, bool)
    if extra:
        mask |= clef_strokes((W, H), tx)
    return mask, (x0, y1)


def morph(mask, r):
    if r == 0:
        return mask
    if r > 0:
        # Only the strokes grow: hairlines are left out of the growth (they keep their width),
        # and the white details of the original (record grooves, slits, key gaps) stay open.
        core = ndimage.distance_transform_edt(mask) > 3
        core = ndimage.distance_transform_edt(~core) <= 3  # opening: thin parts removed
        grown = mask | (ndimage.distance_transform_edt(~core) <= r)
        # Every narrow white passage — counters, slits, grooves and the gaps between neighbouring
        # strokes — keeps its original width, so nothing merges; only open space takes the growth.
        white = ~mask
        rw = r * 1.6 + 2
        white_core = ndimage.distance_transform_edt(white) > rw
        white_core = ndimage.distance_transform_edt(~white_core) <= rw
        thin_white = white & ~white_core
        out = grown & ~thin_white
        # smooth the outline: tiny bumps and nicks from the two rules above are rounded off
        dilated = ndimage.distance_transform_edt(~out) <= 1.8
        closed = ndimage.distance_transform_edt(dilated) > 1.8
        eroded = ndimage.distance_transform_edt(closed) > 1.5
        opened = ndimage.distance_transform_edt(~eroded) <= 1.5
        # the original's own thin white details stay open after smoothing too
        return (opened | (out & mask)) & ~thin_white
    r = -r
    inside = ndimage.distance_transform_edt(mask)
    # the middle line of every band (a pixel at least as deep as all its neighbours)
    ridge = mask & (inside >= ndimage.maximum_filter(inside, 3) - 0.01) & (inside > 0.5)
    # thick parts lose r on each side; every band, however thin, keeps a line of 2·hair around its middle
    hair = morph.hair
    lines = ndimage.distance_transform_edt(~ridge) <= hair
    out = mask & ((inside > r) | lines)
    # fine engraving (record grooves: hair-thin white lines cut into a solid body) is a texture, not
    # strokes: the body is found by closing those lines, and where it is thick the drawing stays as is
    body = ndimage.distance_transform_edt(~mask) <= 4.5
    body = ndimage.distance_transform_edt(body) > 4.5
    out |= mask & (ndimage.distance_transform_edt(body) > r + 4.5)
    # tidy the outline
    dilated = ndimage.distance_transform_edt(~out) <= 1.2
    closed = ndimage.distance_transform_edt(dilated) > 1.2
    return closed & mask


# "Melodix Text": the cut for small sizes. At 12–16 px the engraving (grooves, key gaps, the slits
# between bands) is narrower than a screen pixel and greys the letters into a blot, so every white
# passage narrower than 2·TEXT_GAP is filled; counters, notes, records and big cut-outs stay.
TEXT = os.environ.get("MELODIX_TEXT") == "1"
TEXT_GAP = 20 * PX


RECORDS = {f"mx{c:04X}" for c in (0x4F, 0x6F, 0x41E, 0x43E, 0x42E, 0x44E)}


def solid(mask, record=False):
    white = ~mask
    wide = ndimage.distance_transform_edt(white) > TEXT_GAP
    wide = ndimage.distance_transform_edt(~wide) <= TEXT_GAP
    out = mask | (white & ~wide)
    # a white ring round a black centre (the record's label) stays, however narrow: without it the
    # record O would read as a dot
    labels, n = ndimage.label(white)
    border = set(np.unique(np.concatenate([labels[0], labels[-1], labels[:, 0], labels[:, -1]])))
    for i, sl in enumerate(ndimage.find_objects(labels), 1):
        if i in border or sl is None:
            continue
        part = labels[sl] == i
        if part.sum() > 2000 * PX * PX and (ndimage.binary_fill_holes(part) & ~part).sum() > 300 * PX * PX:
            out[sl] &= ~part
            if record:
                # on records the ring is widened outwards, so at 12–16 px it still reads as O
                ring = ndimage.distance_transform_edt(~(labels == i)) <= 55 * PX
                label_dot = ndimage.binary_fill_holes(labels == i) & ~(labels == i)
                out &= ~(ring & ~label_dot)
    return out


def trace(mask, origin):
    x0, y1 = origin
    path = potrace.Bitmap(~mask).trace(turdsize=8, turnpolicy=potrace.POTRACE_TURNPOLICY_MINORITY, alphamax=1.0, opticurve=True, opttolerance=0.3)
    rec = RecordingPen()
    return path, rec


UA = {0x490: 0x491, 0x404: 0x454, 0x406: 0x456, 0x407: 0x457}


def add_ukrainian(font):
    """Ґ Є І Ї (and small forms) are missing from the large sheet: taken from the Ukrainian sheet's trace."""
    donor = TTFont("out/Melodix-400.ttf")
    dcm, dg, dh = donor.getBestCmap(), donor["glyf"], donor["hmtx"]
    cm = font.getBestCmap()
    order = font.getGlyphOrder()
    for upper, lower in UA.items():
        for cp in (upper, lower):
            if cp in cm and cm[cp].startswith("mx"):
                continue
            src = dcm.get(cp)
            if not src or not src.startswith("mx"):
                continue
            name = f"mx{cp:04X}"
            font["glyf"][name] = dg[src]
            font["hmtx"][name] = dh[src]
            if name not in order:
                order.append(name)
            for t in font["cmap"].tables:
                if t.isUnicode():
                    t.cmap[cp] = name
    font.setGlyphOrder(order)
    font["glyf"].glyphOrder = order
    font["maxp"].numGlyphs = len(order)


CLEF_BOUNDS = (30, -240, 640, 1020)


def clef_strokes(size, tx):
    """The treble clef laid over the S: a hairline stem through the letter, a loop above the capitals
    and a hook with a note head below the line. The S itself stays whole, so it still reads as S."""
    im = Image.new("1", size, 0)
    d = ImageDraw.Draw(im)
    w = max(1, round(24 * PX))
    def box(cx, cy, rx, ry):
        (ax, ay), (bx, by) = tx((cx - rx, cy + ry)), tx((cx + rx, cy - ry))
        return [ax, ay, bx, by]
    # nothing crosses the letter's body (a stem through it reads as $): the loop grows out of the top
    # arc, leaning right, and the tail falls from the bottom arc to the left
    d.line([tx((440, 680)), tx((468, 880))], fill=1, width=w)
    d.ellipse(box(420, 900, 50, 96), outline=1, width=w)                 # the loop above
    d.line([tx((210, 30)), tx((196, -150))], fill=1, width=w)
    d.arc(box(142, -150, 54, 52), 0, 180, fill=1, width=w)               # the hook below
    d.ellipse(box(92, -168, 50, 42), fill=1)                              # the note head ending it
    return np.asarray(im, bool)


def build(weight):
    font = TTFont(SRC)
    add_ukrainian(font)
    glyf, hmtx = font["glyf"], font["hmtx"]
    gs = font.getGlyphSet()
    r = OFFSETS[weight] * PX
    morph.hair = HAIR.get(weight, 0) * PX
    # the field around the glyph must be wider than the gap guard (r·1.6 + 2) or its edge reads as a gap
    pad = int(abs(r) * 3) + 12
    names = sorted({n for n in font.getBestCmap().values() if n.startswith("mx")})
    for name in names:
        res = raster(gs, name, pad, CLEF_BOUNDS if name == "mx0053" else None)
        if res is None:
            continue
        mask, (x0, y1) = res
        out = morph(mask, r)
        if TEXT:
            out = solid(out, name in RECORDS)
        path = potrace.Bitmap(~out).trace(turdsize=8, turnpolicy=potrace.POTRACE_TURNPOLICY_MINORITY, alphamax=1.0, opticurve=True, opttolerance=0.3)
        back = lambda p: ((p.x - pad) / PX + x0, y1 - (p.y - pad) / PX)
        rec = RecordingPen()
        for curve in path:
            rec.moveTo(back(curve.start_point))
            for seg in curve:
                if seg.is_corner:
                    rec.lineTo(back(seg.c))
                    rec.lineTo(back(seg.end_point))
                else:
                    rec.curveTo(back(seg.c1), back(seg.c2), back(seg.end_point))
            rec.closePath()
        def make(source):
            tt = TTGlyphPen(None)
            source.replay(Cu2QuPen(tt, max_err=1.2, reverse_direction=False))
            g = tt.glyph()
            g.recalcBounds(glyf)
            return g
        g = make(rec)
        a = AreaPen()
        chk = RecordingPen()
        g.draw(chk, glyf)
        chk.replay(a)
        if a.value > 0:
            rev = RecordingPen()
            rec.replay(ReverseContourPen(rev))
            g = make(rev)
        glyf[name] = g
        adv, lsb = hmtx[name]
        grow = OFFSETS[weight]
        hmtx[name] = (int(adv + 2 * grow * 0.6), getattr(g, "xMin", lsb))
    style = NAMES[weight]
    font["name"].setName("Melodix", 1, 3, 1, 0x409)
    font["name"].setName(style, 2, 3, 1, 0x409)
    font["name"].setName(f"Melodix {style}", 4, 3, 1, 0x409)
    font["name"].setName(f"Melodix-{style}", 6, 3, 1, 0x409)
    font["OS/2"].usWeightClass = weight
    family, out = ("Melodix Text", "out-text/MelodixText") if TEXT else ("Melodix", "out2/Melodix")
    for i, v in ((1, family), (4, f"{family} {style}"), (6, f"{family.replace(' ', '')}-{style}")):
        font["name"].setName(v, i, 3, 1, 0x409)
    os.makedirs(os.path.dirname(out), exist_ok=True)
    font.save(f"{out}-{weight}.ttf")
    f2 = TTFont(f"{out}-{weight}.ttf")
    opts = subset.Options()
    opts.flavor = "woff2"
    opts.layout_features = ["kern", "liga"]
    opts.name_IDs = ["*"]
    sub = subset.Subsetter(opts)
    cps = list(range(0x20, 0x7F)) + list(range(0xA0, 0x100)) + list(range(0x370, 0x400)) + list(range(0x400, 0x460)) + [0x490, 0x491] + list(range(0x2010, 0x203B)) + [0x20AC, 0x2116, 0x2122, 0x2665] + list(range(0x2669, 0x2670)) + [0x1D11E]
    sub.populate(unicodes=cps)
    sub.subset(f2)
    f2.flavor = "woff2"
    f2.save(f"{out}-{weight}.woff2")
    print("built", weight, len(names))


import sys
for w in (int(a) for a in sys.argv[1:]) if len(sys.argv) > 1 else OFFSETS:
    build(w)
